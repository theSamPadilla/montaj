usage: montaj snapshot [-h] [--cols COLS] [--rows ROWS] [--start START]
                       [--end END] [--frames FRAMES] [--at AT] [--json]
                       [--out PATH] [--quiet]
                       input

positional arguments:
  input            Source video file

options:
  -h, --help       show this help message and exit
  --cols COLS      Grid columns (grid mode, or a sheet of several at)
  --rows ROWS      Grid rows (grid mode)
  --start START    Window start in seconds (grid / all-frames mode)
  --end END        Window end in seconds (grid / all-frames mode)
  --frames FRAMES  Number of frames to sample (grid mode), or 'all' to extract
                   every frame in the window
  --at AT          Seconds. One: a single full-resolution frame (scene
                   continuity, regen stitching). Several (up to 36): one sheet
                   of those times, each tile labelled, cols columns (default
                   up to 6). Not with frames, rows, start, end.
  --json           Output result as JSON
  --out PATH       Output file path
  --quiet          Suppress progress output
