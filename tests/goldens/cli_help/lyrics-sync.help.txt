usage: montaj lyrics-sync [-h] --lyrics LYRICS
                          [--model {large-v3-turbo-q5_0,large-v3-turbo,tiny.en,base.en,small.en,medium.en,tiny,base,small,medium,large,large-v1,large-v2,large-v3}]
                          [--language LANGUAGE] [--start START] [--end END]
                          [--json] [--out PATH] [--quiet]
                          input

positional arguments:
  input                 Vocals WAV from stem_separation (--stems vocals), or
                        any audio/video file

options:
  -h, --help            show this help message and exit
  --lyrics LYRICS       Path to lyrics text file. One phrase per line. Blank
                        lines ignored.
  --model {large-v3-turbo-q5_0,large-v3-turbo,tiny.en,base.en,small.en,medium.en,tiny,base,small,medium,large,large-v1,large-v2,large-v3}
                        Whisper model. Defaults to large-v3-turbo-q5_0; a
                        model that is not installed falls back to one that is.
  --language LANGUAGE   Language code passed to Whisper (default: en)
  --start START         Optional: override start time in seconds (skips auto-
                        detection). Use when auto-detection is off.
  --end END             Optional: override end time in seconds (skips auto-
                        detection).
  --json                Output result as JSON
  --out PATH            Output file path
  --quiet               Suppress progress output
